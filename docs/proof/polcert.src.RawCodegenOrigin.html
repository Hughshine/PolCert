Proof map

This module is a neutral, proof-only trace bridge for standard raw code generation:
  • poly_trace and loop_trace record generated instruction events without introducing state semantics;
  • polyloop_to_loop_trace_reflect reflects LoopGen traces to PolyLoop;
  • generate_loop_many_event_origin recovers the source polyhedral instruction selected by AST generation;
  • schedule elimination and preparation facts then yield complete_generate_many_event_source, the endpoint consumed by ParallelCodegenCorrect.

Neutral traces for PolyLoop

A proof-only trace for the intermediate polyhedral loop language. Points use the polyhedral coordinate order; the operational loop environment is stored in the reverse order.
Record scan_event := {
  se_point : list Z;
  se_instr : Instr.t;
  se_args : list Z
}.

Inductive poly_trace : PolyLoop.poly_stmt → list Z → list scan_event → Prop :=
| PETInstr : ∀ i es env,
    poly_trace
      (PolyLoop.PInstr i es)
      env
      [{| se_point := rev env;
          se_instr := i;
          se_args := map (PolyLoop.eval_affine_expr env) es |}]
| PETSkip : ∀ env,
    poly_trace PolyLoop.PSkip env []
| PETSeq : ∀ s1 s2 env tr1 tr2,
    poly_trace s1 env tr1 →
    poly_trace s2 env tr2 →
    poly_trace (PolyLoop.PSeq s1 s2) env (tr1 ++ tr2)
| PETGuardTrue : ∀ p body env tr,
    in_poly (rev env) p = true →
    poly_trace body env tr →
    poly_trace (PolyLoop.PGuard p body) env tr
| PETGuardFalse : ∀ p body env,
    in_poly (rev env) p = false →
    poly_trace (PolyLoop.PGuard p body) env []
| PETLoop : ∀ p body env lb ub zs trs,
    (∀ x, in_poly (rev (x :: env)) p = true ↔ lb ≤ x < ub) →
    zs = Zrange lb ub →
    Forall2 (fun z tr ⇒ poly_trace body (z :: env) tr) zs trs →
    poly_trace (PolyLoop.PLoop p body) env (concat trs).

Lemma ctx_simplify_resize_equiv :
  ∀ pol ctx n simp point,
    mayReturn (ctx_simplify (resize_poly n pol) ctx) simp →
    Datatypes.length point = n →
    in_poly point ctx = true →
    in_poly point simp = in_poly point pol.
Proof.
  intros pol ctx n simp point Hsimp Hlen Hctx.
  erewrite <- (ctx_simplify_correct _ _ _ Hsimp); [|exact Hctx].
  apply resize_poly_in. exact Hlen.
Qed.

Local Lemma in_poly_cons_preserves_context :
  ∀ ctx env x,
    (poly_nrl ctx ≤ Datatypes.length env)%nat →
    in_poly (rev env) ctx = true →
    in_poly (rev (x :: env)) ctx = true.
Proof.
  intros ctx env x Hctx Henv.
  rewrite <- in_poly_nrlength with (n := Datatypes.length env);
    [|exact Hctx].
  simpl.
  rewrite resize_app by (rewrite rev_length; reflexivity).
  exact Henv.
Qed.

Lemma Forall2_imp_in_left :
  ∀ A B (R S : A → B → Prop) xs ys,
    Forall2 R xs ys →
    (∀ x y, In x xs → R x y → S x y) →
    Forall2 S xs ys.
Proof.
  intros A B R S xs ys Hfor.
  induction Hfor; intro Himp.
  - constructor.
  - constructor.
    + eapply Himp; [left; reflexivity | exact H].
    + apply IHHfor.
      intros x' y' Hin HR.
      eapply Himp; [right; exact Hin | exact HR].
Qed.

polyloop_simplify can remove guards and simplify loop domains, but it does not introduce new execution events. Any trace of the simplified statement is therefore also a trace of the input statement under the simplifier's context.
Lemma polyloop_simplify_trace_reflect :
  ∀ stmt n ctx,
    (poly_nrl ctx ≤ n)%nat →
    ∀ nst,
      mayReturn (PolyLoopSimplifier.polyloop_simplify stmt n ctx) nst →
      ∀ env tr,
        Datatypes.length env = n →
        in_poly (rev env) ctx = true →
        poly_trace nst env tr →
        poly_trace stmt env tr.
Proof.
  induction stmt; intros n ctx Hctx nst Hnst env tr Henv Henvctx Htrace;
    simpl in Hnst.
  - bind_imp_destruct Hnst npol Hnpol.
    bind_imp_destruct Hnst nctx Hnctx.
    bind_imp_destruct Hnst nst1 Hnst1.
    apply mayReturn_pure in Hnst. subst nst.
    inversion Htrace as
      [| | | | |
       p0 body0 env0 lb ub zs trs Hdomain Hrange Htraces];
      subst.
    eapply PETLoop with
      (lb := lb) (ub := ub) (zs := Zrange lb ub) (trs := trs).
    + intros x.
      rewrite <- (Hdomain x).
      assert (Hinctx : in_poly (rev (x :: env)) ctx = true).
      { eapply in_poly_cons_preserves_context; eauto. }
      erewrite <- (ctx_simplify_correct _ _ _ Hnpol); [|exact Hinctx].
      rewrite resize_poly_in; [reflexivity|].
      rewrite rev_length. reflexivity.
    + reflexivity.
    + eapply Forall2_imp_in_left; [exact Htraces|].
      intros x tri Hxin Hbody.
      assert (Hinctx : in_poly (rev (x :: env)) ctx = true).
      { eapply in_poly_cons_preserves_context; eauto. }
      assert (Hnctxnrl :
        (poly_nrl nctx ≤ S (Datatypes.length env))%nat).
      { eapply poly_inter_nrl; [apply resize_poly_nrl| |exact Hnctx]; lia. }
      eapply IHstmt with (n := S (Datatypes.length env)) (ctx := nctx).
      × exact Hnctxnrl.
      × exact Hnst1.
      × reflexivity.
      × erewrite poly_inter_def, poly_inter_pure_def, resize_poly_in;
          [|rewrite rev_length; reflexivity|exact Hnctx].
        reflect. simpl. split; [|exact Hinctx].
        change (in_poly (rev (x :: env)) p = true).
        rewrite <- (resize_poly_in
          (rev (x :: env)) p (S (Datatypes.length env)))
          by (rewrite rev_length; reflexivity).
        erewrite (ctx_simplify_correct _ _ _ Hnpol); [|exact Hinctx].
        apply (proj2 (Hdomain x)).
        rewrite <- Zrange_in. exact Hxin.
      × exact Hbody.
  - apply mayReturn_pure in Hnst. subst nst. exact Htrace.
  - apply mayReturn_pure in Hnst. subst nst. exact Htrace.
  - bind_imp_destruct Hnst s1 Hs1.
    bind_imp_destruct Hnst s2 Hs2.
    apply mayReturn_pure in Hnst. subst nst.
    inversion Htrace; subst.
    constructor.
    + eapply IHstmt1; eauto.
    + eapply IHstmt2; eauto.
  - bind_imp_destruct Hnst npol Hnpol.
    bind_imp_destruct Hnst nctx Hnctx.
    bind_imp_destruct Hnst nst1 Hnst1.
    apply mayReturn_pure in Hnst. subst nst.
    inversion Htrace as
      [| | |
       p0 body0 env0 tr0 Htest Hbody
       | p0 body0 env0 Htest
       |];
      subst.
    + apply PETGuardTrue.
      × rewrite (ctx_simplify_resize_equiv
          p ctx (Datatypes.length env) npol (rev env)
          Hnpol (eq_trans (rev_length env) eq_refl) Henvctx) in Htest.
        exact Htest.
      × assert (Hnctxnrl :
          (poly_nrl nctx ≤ Datatypes.length env)%nat).
        { eapply poly_inter_nrl; [apply resize_poly_nrl|exact Hctx|exact Hnctx]. }
        eapply IHstmt with (n := Datatypes.length env) (ctx := nctx).
        -- exact Hnctxnrl.
        -- exact Hnst1.
        -- reflexivity.
        -- erewrite poly_inter_def, poly_inter_pure_def, resize_poly_in;
             [|rewrite rev_length; reflexivity|exact Hnctx].
           reflect. simpl. split; [|exact Henvctx].
           rewrite (ctx_simplify_resize_equiv
             p ctx (Datatypes.length env) npol (rev env)
             Hnpol (rev_length env) Henvctx) in Htest.
           exact Htest.
        -- exact Hbody.
    + apply PETGuardFalse.
      rewrite (ctx_simplify_resize_equiv
        p ctx (Datatypes.length env) npol (rev env)
        Hnpol (eq_trans (rev_length env) eq_refl) Henvctx) in Htest.
      exact Htest.
Qed.

Inductive loop_trace : Loop.stmt → list Z → list scan_event → Prop :=
| LETInstr : ∀ i es env,
    loop_trace
      (Loop.Instr i es)
      env
      [{| se_point := rev env;
          se_instr := i;
          se_args := map (Loop.eval_expr env) es |}]
| LETSeq : ∀ ss env tr,
    loop_traces ss env tr →
    loop_trace (Loop.Seq ss) env tr
| LETGuardTrue : ∀ tst body env tr,
    Loop.eval_test env tst = true →
    loop_trace body env tr →
    loop_trace (Loop.Guard tst body) env tr
| LETGuardFalse : ∀ tst body env,
    Loop.eval_test env tst = false →
    loop_trace (Loop.Guard tst body) env []
| LETLoop : ∀ lb ub body env zs trs,
    zs = Zrange (Loop.eval_expr env lb) (Loop.eval_expr env ub) →
    Forall2 (fun z tr ⇒ loop_trace body (z :: env) tr) zs trs →
    loop_trace (Loop.Loop lb ub body) env (concat trs)
with loop_traces : Loop.stmt_list → list Z → list scan_event → Prop :=
| LETNil : ∀ env,
    loop_traces Loop.SNil env []
| LETCons : ∀ s ss env tr1 tr2,
    loop_trace s env tr1 →
    loop_traces ss env tr2 →
    loop_traces (Loop.SCons s ss) env (tr1 ++ tr2).

Definition same_effect
    (ip1 ip2 : PolyLang.InstrPoint) : Prop :=
  ip1.(PolyLang.ip_instruction) = ip2.(PolyLang.ip_instruction) ∧
  affine_product ip1.(PolyLang.ip_transformation) ip1.(PolyLang.ip_index) =
  affine_product ip2.(PolyLang.ip_transformation) ip2.(PolyLang.ip_index).

Lemma instr_point_sema_same_effect_iff :
  ∀ ip1 ip2 st1 st2,
    same_effect ip1 ip2 →
    ILSema.instr_point_sema ip1 st1 st2 ↔
    ILSema.instr_point_sema ip2 st1 st2.
Proof.
  intros ip1 ip2 st1 st2 [Hinstr Hargs].
  split; intro Hsem; inversion Hsem as [wcs rcs Hstep]; subst.
  - econstructor.
    rewrite <- Hinstr, <- Hargs.
    exact Hstep.
  - econstructor.
    rewrite Hinstr, Hargs.
    exact Hstep.
Qed.

Lemma same_effect_symm :
  ∀ ip1 ip2,
    same_effect ip1 ip2 →
    same_effect ip2 ip1.
Proof.
  intros ip1 ip2 [Hinstr Hargs].
  split; symmetry; assumption.
Qed.

Lemma Permutable_same_effect :
  ∀ ip1 ip1' ip2 ip2',
    same_effect ip1 ip1' →
    same_effect ip2 ip2' →
    ILSema.Permutable ip1 ip2 →
    ILSema.Permutable ip1' ip2'.
Proof.
  intros ip1 ip1' ip2 ip2' Heffect1 Heffect2 Hperm.
  unfold ILSema.Permutable in ×.
  intros st1 Hnonalias.
  specialize (Hperm st1 Hnonalias).
  destruct Hperm as [Hforward Hbackward].
  split.
  - intros st2 st3 Hstep1 Hstep2.
    apply (proj2
      (instr_point_sema_same_effect_iff ip1 ip1' st1 st2 Heffect1))
      in Hstep1.
    apply (proj2
      (instr_point_sema_same_effect_iff ip2 ip2' st2 st3 Heffect2))
      in Hstep2.
    destruct (Hforward _ _ Hstep1 Hstep2)
      as [st2' [st3' [Hstep2' [Hstep1' Heq]]]].
    ∃ st2', st3'.
    repeat split; auto.
    + apply (proj1
        (instr_point_sema_same_effect_iff ip2 ip2' st1 st2' Heffect2)).
      exact Hstep2'.
    + apply (proj1
        (instr_point_sema_same_effect_iff ip1 ip1' st2' st3' Heffect1)).
      exact Hstep1'.
  - intros st2 st3 Hstep2 Hstep1.
    apply (proj2
      (instr_point_sema_same_effect_iff ip2 ip2' st1 st2 Heffect2))
      in Hstep2.
    apply (proj2
      (instr_point_sema_same_effect_iff ip1 ip1' st2 st3 Heffect1))
      in Hstep1.
    destruct (Hbackward _ _ Hstep2 Hstep1)
      as [st2' [st3' [Hstep1' [Hstep2' Heq]]]].
    ∃ st2', st3'.
    repeat split; auto.
    + apply (proj1
        (instr_point_sema_same_effect_iff ip1 ip1' st1 st2' Heffect1)).
      exact Hstep1'.
    + apply (proj1
        (instr_point_sema_same_effect_iff ip2 ip2' st2' st3' Heffect2)).
      exact Hstep2'.
Qed.

Lemma Forall2_in_right :
  ∀ A B (R : A → B → Prop) xs ys y,
    Forall2 R xs ys →
    In y ys →
    ∃ x, In x xs ∧ R x y.
Proof.
  intros A B R xs ys y Hfor.
  induction Hfor; intro Hin.
  - inversion Hin.
  - simpl in Hin.
    destruct Hin as [Heq | Hin].
    + subst y. ∃ x. split; [left; reflexivity | assumption].
    + destruct (IHHfor Hin) as [x' [Hin' HR]].
      ∃ x'. split; [right; exact Hin' | exact HR].
Qed.

Lemma poly_trace_event_extends_env :
  ∀ s env tr ev,
    poly_trace s env tr →
    In ev tr →
    ∃ suffix, se_point ev = rev env ++ suffix.
Proof.
  induction s; intros env tr ev Htrace Hin;
    inversion Htrace as
      [instr0 exprs0 env0
      |env0
      |s1' s2' env0 tr1 tr2 Htrace1 Htrace2
      |p' body env0 tr0 Htest Hbody
      |p' body env0 Htest
      |p' body env0 lb ub zs trs Hdomain Hrange Htraces];
    subst.
  - apply in_concat in Hin.
    destruct Hin as [tr0 [Htr0 Hin]].
    destruct (Forall2_in_right _ _ _ _ _ _ Htraces Htr0)
      as [z [_ Hz]].
    destruct (IHs (z :: env) tr0 ev Hz Hin) as [suffix Hevent].
    ∃ (z :: suffix).
    simpl in Hevent.
    rewrite Hevent.
    symmetry.
    exact (app_assoc (rev env) [z] suffix).
  - simpl in Hin.
    destruct Hin as [Heq | Hin]; [subst ev | contradiction].
    ∃ []. now rewrite app_nil_r.
  - contradiction.
  - apply in_app_iff in Hin.
    destruct Hin as [Hin | Hin].
    + eapply IHs1; eauto.
    + eapply IHs2; eauto.
  - eapply IHs; eauto.
  - contradiction.
Qed.

Schedule elimination inserts exactly width coordinates immediately after the symbolic environment.
Lemma pi_elim_schedule_inserted_coordinates :
  ∀ width env_dim pi envv sched_coords source_coords,
    (Datatypes.length pi.(PolyLang.pi_schedule) ≤ width)%nat →
    Datatypes.length envv = env_dim →
    Datatypes.length sched_coords = width →
    in_poly
      (envv ++ sched_coords ++ source_coords)
      (PolyLang.pi_elim_schedule width env_dim pi).(PolyLang.pi_poly) = true →
    sched_coords =
      resize width
        (affine_product pi.(PolyLang.pi_schedule)
          (envv ++ source_coords)).
Proof.
  intros width env_dim pi envv sched_coords source_coords
    Hschedule_width Henv_width Hcoords_width Hdomain.
  unfold PolyLang.pi_elim_schedule in Hdomain.
  simpl in Hdomain.
  rewrite in_poly_app in Hdomain.
  apply andb_true_iff in Hdomain.
  destruct Hdomain as [Hschedule_domain _].
  rewrite
    (PolyLang.make_sched_poly_correct
       pi.(PolyLang.pi_schedule) width env_dim Hschedule_width
       envv sched_coords source_coords Henv_width Hcoords_width)
    in Hschedule_domain.
  pose proof
    (resize_eq_compat
       width sched_coords
       (affine_product pi.(PolyLang.pi_schedule)
          (envv ++ source_coords))
       Hschedule_domain) as Hresized.
  rewrite resize_length_eq in Hresized by exact Hcoords_width.
  exact Hresized.
Qed.

Lemma loop_trace_make_guard_inv :
  ∀ tst body env tr,
    loop_trace (Loop.make_guard tst body) env tr →
    (Loop.eval_test env tst = true ∧ loop_trace body env tr) ∨
    (Loop.eval_test env tst = false ∧ tr = []).
Proof.
  intros tst body env tr Htrace.
  unfold Loop.make_guard in Htrace.
  destruct tst as [e1 e2 | e1 e2 | t1 t2 | t1 t2 | t | b];
    simpl in Htrace |- ×.
  - inversion Htrace; subst; [left | right]; auto.
  - inversion Htrace; subst; [left | right]; auto.
  - inversion Htrace; subst; [left | right]; auto.
  - inversion Htrace; subst; [left | right]; auto.
  - inversion Htrace; subst; [left | right]; auto.
  - destruct b.
    + left. auto.
    + inversion Htrace as [|ss env0 tr0 Hnil| | |]; subst.
      inversion Hnil; subst.
      right. split; reflexivity.
Qed.

Lemma loop_trace_make_let_inv :
  ∀ value body env tr,
    loop_trace (Loop.make_let value body) env tr →
    loop_trace body (Loop.eval_expr env value :: env) tr.
Proof.
  intros value body env tr Htrace.
  unfold Loop.make_let in Htrace.
  inversion Htrace as
    [| | | |lb ub body0 env0 zs trs Hrange Hiterations]; subst.
  simpl in Hiterations.
  rewrite Zrange_single in Hiterations.
  inversion Hiterations as
    [|z zs0 tr0 trs0 Hbody Hremaining]; subst.
  inversion Hremaining; subst.
  simpl.
  now rewrite app_nil_r.
Qed.

Local Lemma empty_scan_trace_witness :

    + lia.
  - split.

Qed.

Trace counterpart of LoopGenCore.scan_dimension_sem. In particular, the trace iterations are tied to the same interval that characterizes the scanned polyhedron, rather than to an arbitrary list of coordinates.
Lemma scan_dimension_trace_inv :
  ∀ n inner pol out,
    mayReturn (LoopGenCore.scan_dimension n inner pol) out →
    ∀ env tr,
      length env = n →
      loop_trace out env tr →
      ∃ lb ub zs trs,
        (∀ x,
          in_poly (rev (x :: env)) pol = true ↔ lb ≤ x < ub) ∧
        zs = Zrange lb ub ∧
        tr = concat trs ∧
        Forall2 (fun z tri ⇒ loop_trace inner (z :: env) tri) zs trs.
Proof.
  intros n inner pol out Hout env tr Henv.
  unfold LoopGenCore.scan_dimension in Hout.
  destruct (find_eq n pol) as [c |] eqn:Hfindeq.
  - destruct (LoopGenCore.solve_eq n c) as [result test] eqn:Hsolve.
    apply mayReturn_pure in Hout. subst out.
    set (test1 :=
      Loop.make_and test
        (Loop.and_all
          (map
             (fun c1 ⇒
                LoopGenCore.make_affine_test n
                  (make_constraint_with_eq n c c1))
             (filter (fun c1 ⇒ negb (nth n (fst c1) 0 =? 0)) pol) ++
           map (LoopGenCore.make_affine_test n)
             (filter (fun c1 ⇒ nth n (fst c1) 0 =? 0) pol)))).
    assert (Hcnth : 0 < nth n (fst c) 0) by
      (eapply find_eq_nth; eauto).
    intro Htrace.
    destruct (loop_trace_make_guard_inv test1
      (Loop.make_let result inner) env tr Htrace)
      as [[Htest1 Hbody] | [Htest1 Htr]].
    + pose proof (loop_trace_make_let_inv result inner env tr Hbody)
        as Hinner.
      ∃ (Loop.eval_expr env result).
      ∃ (Loop.eval_expr env (Loop.Sum result (Loop.Constant 1))).
      ∃ [Loop.eval_expr env result].
      ∃ [tr].
      split.
      × intros x. simpl.
        unfold test1 in Htest1.
        rewrite Loop.make_and_correct in Htest1; reflect.
        rewrite Loop.and_all_correct in Htest1.
        destruct Htest1 as [Htest Hcstr].
        rewrite forallb_app, andb_true_iff in Hcstr.
        destruct Hcstr as [Hcstr1 Hcstr2].
        transitivity
          (Loop.eval_expr env (fst (LoopGenCore.solve_eq n c)) = x ∧
           Loop.eval_test env (snd (LoopGenCore.solve_eq n c)) = true).
        -- rewrite LoopGenCore.solve_eq_correct by (auto || lia).
           split.
           ++ intros H. unfold in_poly in H.
              rewrite forallb_forall in H.
              eapply find_eq_correct_1; eauto.
           ++ intros H. unfold in_poly; rewrite forallb_forall.
              intros c1 Hc1in.
              rewrite forallb_map, forallb_forall in Hcstr1, Hcstr2.
              specialize (Hcstr1 c1). specialize (Hcstr2 c1).
              rewrite filter_In,
                LoopGenCore.make_affine_test_correct in Hcstr1, Hcstr2 by auto.
              destruct (nth n (fst c1) 0 =? 0) eqn:Hc1nth; reflect.
              ** erewrite LoopGenCore.satisfies_constraint_nth_zero_eq;
                   rewrite ?rev_length; eauto.
              ** rewrite <- make_constraint_with_eq_correct_1

Inductive poly_event_origin :
    PolyLoop.poly_stmt → list Z → scan_event → Prop :=
| PEOInstr : ∀ i es env,
    poly_event_origin
      (PolyLoop.PInstr i es) env
      {| se_point := rev env;
         se_instr := i;
         se_args := map (PolyLoop.eval_affine_expr env) es |}
| PEOSeqLeft : ∀ s1 s2 env ev,
    poly_event_origin s1 env ev →
    poly_event_origin (PolyLoop.PSeq s1 s2) env ev
| PEOSeqRight : ∀ s1 s2 env ev,
    poly_event_origin s2 env ev →
    poly_event_origin (PolyLoop.PSeq s1 s2) env ev
| PEOGuard : ∀ pol body env ev,
    in_poly (rev env) pol = true →
    poly_event_origin body env ev →
    poly_event_origin (PolyLoop.PGuard pol body) env ev
| PEOLoop : ∀ pol body env x ev,
    in_poly (rev (x :: env)) pol = true →
    poly_event_origin body (x :: env) ev →
    poly_event_origin (PolyLoop.PLoop pol body) env ev.

Theorem generate_loop_event_origin :
  ∀ d n pi pstmt env ev,
    (d ≤ n)%nat →
    mayReturn (ASTGenCore.generate_loop d n pi) pstmt →
    length env = (n - d)%nat →
    (∀ c, In c pi.(PolyLang.pi_poly) →
      fst c =v= resize n (fst c)) →
    project_invariant (n - d)%nat pi.(PolyLang.pi_poly) (rev env) →
    poly_event_origin pstmt env ev →
    generated_pi_event_origin n pi ev.
Proof.
  induction d;
    intros n pi pstmt env ev Hdn Hgen Henv Hcols Hinv Horigin;
    simpl in Hgen.
  - apply mayReturn_pure in Hgen. subst pstmt.
    inversion Horigin as [instr0 exprs0 env0| | | |]; subst.
    rewrite Nat.sub_0_r in Henv, Hinv.
    constructor.
    + simpl. rewrite rev_length. exact Henv.
    + simpl. eapply project_id; eauto.
    + simpl. reflexivity.
    + unfold PolyLang.current_src_args_in_dim,
        PolyLang.current_src_args_at.
      simpl.
      rewrite map_map.
      apply map_ext.
      intros [v c].
      unfold PolyLoop.eval_affine_expr. simpl.
      rewrite Z.div_1_r.
      reflexivity.
  - bind_imp_destruct Hgen proj Hproj.
    bind_imp_destruct Hgen inner Hinner.
    apply mayReturn_pure in Hgen. subst pstmt.
    inversion Horigin as
      [| | | |pol0 body0 env0 x0 ev0 Hprojected Hinner_origin]; subst.
    eapply IHd with (env := x0 :: env).
    + lia.
    + exact Hinner.
    + simpl. lia.
    + exact Hcols.
    + replace (n - d)%nat with (S (n - S d))%nat in × by lia.
      eapply project_next_r_inclusion; [|exact Hproj|].
      × simpl.
        rewrite project_invariant_resize, resize_app

        rewrite forallb_forall in Hprojected.

Definition drop_schedule_coords
    (env_dim width : nat) (current : list Z) : list Z :=
  resize env_dim current ++ skipn (width + env_dim)%nat current.

Lemma drop_schedule_coords_length :
  ∀ env_dim dim width current,
    length current = (dim + width)%nat →
    (env_dim ≤ dim)%nat →
    length (drop_schedule_coords env_dim width current) = dim.
Proof.
  intros env_dim dim width current Hlength Henvdim.
  unfold drop_schedule_coords.
  rewrite app_length, resize_length, skipn_length, Hlength.
  lia.
Qed.

Lemma drop_schedule_coords_prefix :
  ∀ env_dim width current,
    firstn env_dim (drop_schedule_coords env_dim width current) =
    resize env_dim current.
Proof.
  intros env_dim width current.
  unfold drop_schedule_coords.
  rewrite firstn_app, resize_length, Nat.sub_diag.
  simpl. rewrite firstn_all2 by (rewrite resize_length; lia).
  now rewrite app_nil_r.
Qed.

Lemma pi_elim_schedule_domain_source :
  ∀ width env_dim pi current,
    in_poly current
      (PolyLang.pi_elim_schedule width env_dim pi).(PolyLang.pi_poly) = true →
    in_poly (drop_schedule_coords env_dim width current)
      pi.(PolyLang.pi_poly) = true.
Proof.
  intros width env_dim pi current Hdomain.
  unfold PolyLang.pi_elim_schedule in Hdomain.
  simpl in Hdomain.
  rewrite in_poly_app in Hdomain.
  apply andb_true_iff in Hdomain.
  destruct Hdomain as [_ Hsource].
  unfold in_poly in ×.
  rewrite forallb_map, forallb_forall in Hsource.
  rewrite forallb_forall.
  intros c Hcin.
  specialize (Hsource c Hcin).
  unfold satisfies_constraint, PolyLang.insert_zeros_constraint in Hsource.
  simpl in Hsource.
  rewrite dot_product_commutative in Hsource.
  rewrite PolyLang.insert_zeros_product_skipn in Hsource.
  unfold satisfies_constraint, drop_schedule_coords.
  rewrite dot_product_commutative.
  exact Hsource.
Qed.

Lemma pi_elim_schedule_event_coordinates :
  ∀ width env_dim pi current,
    (length pi.(PolyLang.pi_schedule) ≤ width)%nat →
    in_poly current
      (PolyLang.pi_elim_schedule width env_dim pi).(PolyLang.pi_poly) = true →
    resize width (skipn env_dim current) =
      resize width
        (affine_product pi.(PolyLang.pi_schedule)
          (drop_schedule_coords env_dim width current)).
Proof.
  intros width env_dim pi current Hwidth Hdomain.
  pose proof (PolyLang.split3_eq env_dim width current) as Hsplit.
  assert (Hdomain_split :
    in_poly
      (resize env_dim current ++
       resize width (skipn env_dim current) ++
       skipn (width + env_dim)%nat current)
      (PolyLang.pi_elim_schedule width env_dim pi).(PolyLang.pi_poly) = true).
  {
    rewrite Hsplit. exact Hdomain.
  }
  pose proof
    (pi_elim_schedule_inserted_coordinates
      width env_dim pi
      (resize env_dim current)
      (resize width (skipn env_dim current))
      (skipn (width + env_dim)%nat current)
      Hwidth (resize_length env_dim current)
      (resize_length width (skipn env_dim current))
      Hdomain_split) as Hcoords.
  unfold drop_schedule_coords.
  exact Hcoords.
Qed.

Definition prepared_source_event
    (ppis : list PolyLang.PolyInstr)
    (env_dim dim width : nat) (ev : scan_event) : Prop :=
  ∃ source_nth source_pi,
    nth_error ppis source_nth = Some source_pi ∧
    length (drop_schedule_coords env_dim width ev.(se_point)) = dim ∧
    in_poly (drop_schedule_coords env_dim width ev.(se_point))
      source_pi.(PolyLang.pi_poly) = true ∧
    ev.(se_instr) = source_pi.(PolyLang.pi_instr) ∧
    ev.(se_args) =
      PolyLang.current_src_args_in_dim dim source_pi
        (drop_schedule_coords env_dim width ev.(se_point)) ∧
    firstn env_dim (drop_schedule_coords env_dim width ev.(se_point)) =
      resize env_dim ev.(se_point) ∧
    resize width (skipn env_dim ev.(se_point)) =
      resize width
        (affine_product source_pi.(PolyLang.pi_schedule)
          (drop_schedule_coords env_dim width ev.(se_point))).

Theorem elim_schedule_event_source :
  ∀ ppis env_dim dim width ev,
    (env_dim ≤ dim)%nat →
    (∀ pi, In pi ppis →
      PolyLang.current_env_dim_in_dim dim pi.(PolyLang.pi_point_witness) =
        env_dim) →
    (∀ pi, In pi ppis →
      (length pi.(PolyLang.pi_schedule) ≤ width)%nat) →
    generated_program_event_origin
      (dim + width)%nat (PolyLang.elim_schedule width env_dim ppis) ev →
    prepared_source_event ppis env_dim dim width ev.
Proof.
  intros ppis env_dim dim width ev Henvdim Hcurrent_env Hwidth Horigin.
  destruct Horigin as [m [epi [Hepi Hgenerated]]].
  unfold PolyLang.elim_schedule in Hepi.
  rewrite nth_error_map_iff in Hepi.
  destruct Hepi as [pi [Hpi Helim]].
  subst epi.
  assert (Hpiin : In pi ppis) by (eapply nth_error_In; eauto).
  destruct Hgenerated as [Hpoint_length Hdomain Hinstr Hargs].
  ∃ m, pi.
  split; [exact Hpi|].
  repeat split.
  - eapply drop_schedule_coords_length; eauto.
  - eapply pi_elim_schedule_domain_source. exact Hdomain.
  - rewrite Hinstr. reflexivity.
  -
    rewrite Hargs.
    rewrite PolyLang.current_src_args_in_dim_pi_elim_schedule_resize.
    rewrite (Hcurrent_env pi Hpiin).
    unfold drop_schedule_coords.
    reflexivity.
  - apply drop_schedule_coords_prefix.
  -
    eapply pi_elim_schedule_event_coordinates; eauto.
Qed.

Theorem complete_generate_many_event_source :
  ∀ env_dim dim pis out env tr ev,
    (env_dim ≤ dim)%nat →
    mayReturn (CodeGenCore.complete_generate_many env_dim dim pis) out →
    Datatypes.length env = env_dim →
    ASTGenCore.pis_have_dimension pis dim →
    (∀ pi, In pi pis →
      PolyLang.current_env_dim_in_dim dim
        pi.(PolyLang.pi_point_witness) = env_dim) →
    (∀ pi, In pi pis →
      (poly_nrl pi.(PolyLang.pi_schedule) ≤ dim)%nat) →
    loop_trace out env tr →
    In ev tr →
    prepared_source_event pis env_dim dim
      (list_max
        (map (fun pi ⇒ Datatypes.length pi.(PolyLang.pi_schedule)) pis))
      ev.
Proof.
  intros env_dim dim pis out env tr ev Henvdim Hgen Henv Hpis
    Hpis_envdim Hpis_schedule Htrace Hevent.
  unfold CodeGenCore.complete_generate_many in Hgen.
  set (width :=
    list_max
      (map (fun pi ⇒ Datatypes.length pi.(PolyLang.pi_schedule)) pis))
    in ×.
  unfold CodeGenCore.complete_generate_lex_many in Hgen.
  bind_imp_destruct Hgen pstmt Hpstmt.
  bind_imp_destruct Hgen simp Hsimp.
  assert (Helim_dim :
    ASTGenCore.pis_have_dimension
      (PolyLang.elim_schedule width env_dim pis) (dim + width)%nat).
  {
    unfold ASTGenCore.pis_have_dimension, PolyLang.elim_schedule in ×.
    rewrite forallb_map, forallb_forall in ×.
    intros pi Hpi.
    specialize (Hpis pi Hpi).
    reflect.
    rewrite PolyLang.pi_elim_schedule_nrl by
      (apply list_max_ge; rewrite in_map_iff; ∃ pi; auto).
    specialize (Hpis_schedule pi Hpi).
    lia.
  }
  assert (Hsimp_trace : poly_trace simp env tr).
  {
    eapply polyloop_to_loop_trace_reflect.
    - exact Hgen.
    - lia.
    - exact Htrace.
  }
  assert (Hpolytrace : poly_trace pstmt env tr).
  {
    eapply polyloop_simplify_trace_reflect with
      (n := env_dim) (ctx := nil) (nst := simp).
    - unfold poly_nrl. simpl. lia.
    - replace env_dim with
        ((dim + width) - (dim + width - env_dim))%nat by lia.
      exact Hsimp.
    - exact Henv.
    - reflexivity.
    - exact Hsimp_trace.
  }
  assert (Horigin :
    generated_program_event_origin
      (dim + width)%nat (PolyLang.elim_schedule width env_dim pis) ev).

      (dim + width - env_dim)%nat (dim + width)%nat
      (PolyLang.elim_schedule width env_dim pis) pstmt env ev).
    - lia.
    - exact Hpstmt.
    - lia.
    - exact Helim_dim.
    - eapply poly_trace_event_origin; eauto.
  }
  eapply elim_schedule_event_source; eauto.
  intros pi Hpi.
  apply list_max_ge.
  rewrite in_map_iff.
  ∃ pi. auto.
Qed.
