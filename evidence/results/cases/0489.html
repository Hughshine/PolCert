Source Loop Program

context(N);

for i0 in range(0, N) {
  x[i0] = (a[i0] + b[i0]);
}
s = 0.0;
for i0 in range(0, N) {
  s = (s + (x[i0] * 3));
}

Accepted Optimized Loop Program

context(N);

s = 0.0;
if (1 <= N) {
  for i0 in range(0, N) {
    x[i0] = (a[i0] + b[i0]);
    s = (s + (x[i0] * 3));
  }
}
