Why this test matters
The test distinguishes proved block unrolling and local jamming from a producer option that has no checked structural effect.

Accepted Program

A[0][0] = 1;
for i0 in range(1, 199) {
  if (i0 <= 99) {
    A[i0][0] = 1;
  }
  for i1 in range(max(1, (i0 + -99)), min((i0 + 1), 100)) {
    for i2 in range((i0 + (-1 * i1)), (i0 + ((-1 * i1) + 1))) {
      A[i2][i1] = 1;
    }
    for i2 in range((i0 + (-1 * i1)), (i0 + ((-1 * i1) + 1))) {
      for i3 in range((i1 + -1), i1) {
        B[i2][i3] = (A[(i2 + 1)][i3] + A[i2][(i3 + 1)]);
      }
    }
  }
  if (100 <= i0) {
    for i1 in range((i0 + -100), (i0 + -99)) {
      B[i1][99] = (A[(i1 + 1)][99] + A[i1][100]);
    }
  }
}
B[99][99] = (A[100][99] + A[99][100]);
