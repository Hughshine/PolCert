What This Case Checks

Compares the input Loop program with the output of the checked unroll-and-jam postpass.

Expected outcome

Why this test matters
The test distinguishes proved block unrolling and local jamming from a producer option that has no checked structural effect.

Before Program

context(M, N, K);

for i in range(0, M) {
  for j in range(0, N) {
    for k in range(0, K) {
      C[i][j] = ((beta * C[i][j]) + ((alpha * A[i][k]) * B[k][j]));
    }
  }
}

Accepted Program

context(M, N, K);

if ((1 <= N && 1 <= K)) {
  if (1 <= M) {
    for i0 in range(0, (M / 4)) {
      for i1 in range(0, N) {
        for i2 in range(0, K) {
          C[(4 * i0)][i1] = ((beta * C[(4 * i0)][i1]) + ((alpha * A[(4 * i0)][i2]) * B[i2][i1]));
          C[((4 * i0) + 1)][i1] = ((beta * C[((4 * i0) + 1)][i1]) + ((alpha * A[((4 * i0) + 1)][i2]) * B[i2][i1]));
        }
        for i2 in range(0, K) {
          C[((4 * i0) + 2)][i1] = ((beta * C[((4 * i0) + 2)][i1]) + ((alpha * A[((4 * i0) + 2)][i2]) * B[i2][i1]));
        }
        for i2 in range(0, K) {
          C[((4 * i0) + 3)][i1] = ((beta * C[((4 * i0) + 3)][i1]) + ((alpha * A[((4 * i0) + 3)][i2]) * B[i2][i1]));
        }
      }
    }
    for i0 in range((4 * (M / 4)), M) {
      for i1 in range(0, N) {
        for i2 in range(0, K) {
          C[i0][i1] = ((beta * C[i0][i1]) + ((alpha * A[i0][i2]) * B[i2][i1]));
        }
      }
    }
  }
}
