Accepted Optimized Loop Program

context(n);

if (1 <= n) {
  for i0 in range(0, ((n + 31) / 32)) {
    for i1 in range(0, ((n + 31) / 32)) {
      for i2 in range(max((32 * i0), 0), min(((32 * i0) + 32), n)) {
        if (i1 <= 0) {
          a[i2][0] = (b[i2][-1] + a[i2][-1]);
        }
        for i3 in range(max(1, (32 * i1)), min(((32 * i1) + 32), n)) {
          for i4 in range((i3 + -1), i3) {
            b[i2][i4] = (a[(i2 + -1)][(i4 + 1)] + b[(i2 + -1)][i4]);
          }
          a[i2][i3] = (b[i2][(i3 + -1)] + a[i2][(i3 + -1)]);
        }
        if ((n + (-32 * i1)) <= 31) {
          for i3 in range((n + -1), n) {
            b[i2][i3] = (a[(i2 + -1)][(i3 + 1)] + b[(i2 + -1)][i3]);
          }
        }
      }
    }
    if ((n % 32) == 0) {
      for i1 in range((32 * i0), ((32 * i0) + 32)) {
        if ((n % 32) == 0) {
          for i2 in range((n + -1), n) {
            b[i1][i2] = (a[(i1 + -1)][(i2 + 1)] + b[(i1 + -1)][i2]);
          }
        }
      }
    }
  }
}
