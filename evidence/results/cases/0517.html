Source Loop Program

context(N, M);

for i0 in range(0, N) {
  for i1 in range(0, M) {
    s = (s + a[i0][i1]);
  }
}

Accepted Optimized Loop Program

context(N, M);

if (1 <= M) {
  for i0 in range(0, N) {
    for i1 in range(0, M) {
      s = (s + a[i0][i1]);
    }
  }
}
