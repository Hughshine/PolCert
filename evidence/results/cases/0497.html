Accepted Optimized Loop Program

context(N, M);

if ((1 <= N && 1 <= M)) {
  for i0 in range(0, ((N + 31) / 32)) {
    for i1 in range(0, ((M + 31) / 32)) {
      for i2 in range(max(0, (32 * i0)), min(N, ((32 * i0) + 32))) {
        for i3 in range(max(0, (32 * i1)), min(M, ((32 * i1) + 32))) {
          a[i2] = (a[i2] + (b[i3] * c[i3][i2]));
        }
      }
    }
  }
}
