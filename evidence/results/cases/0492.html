Accepted Optimized Loop Program

context(N);

if (2 <= N) {
  for i0 in range(0, ((N + 31) / 32)) {
    for i1 in range(0, ((N + 31) / 32)) {
      for i2 in range(max(0, (32 * i0)), min(N, ((32 * i0) + 32))) {
        if (i1 <= 0) {
          a[(2 * i2)] = (a[(2 * i2)] + b[0]);
        }
        for i3 in range(max(1, (32 * i1)), min(N, ((32 * i1) + 32))) {
          a[(2 * i2)] = (a[(2 * i2)] + b[i3]);
          a[((2 * i2) + 1)] = (a[((2 * i2) + 1)] + b[i3]);
        }
      }
    }
  }
}
if ((N <= 1 && 1 <= N)) {
  a[0] = (a[0] + b[0]);
}
